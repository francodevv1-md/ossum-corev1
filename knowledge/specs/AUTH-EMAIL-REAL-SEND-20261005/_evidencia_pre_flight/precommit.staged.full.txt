+  const handleRemoveEmail = (type: "to" | "cc", emailToRemove: string) => {

+
+  const handleRemoveAttachment = (index: number) => {

+    institutionName: surgery.institution || "ÔÇö",
+    surgeryDate: surgery.date || "ÔÇö",

+    prestadorEmail: "ventas@districorr.com.ar",
+    prestadorAddress: "ÔÇö",

+        code: "SHAV",

+      },
+      {
+        code: "SISTI",
+        description: "Sistema de Titanio Interferencial para plastia de LCA - Nacional",

+        date: surgery.date || new Date().toLocaleDateString("es-AR"),
+        quantity: 1,
+      },
+    ],
+    notes,
+    ...authorizationData,
+  }
+

+  const previewHtml =
+    mode === "authorization"
+      ? generateAuthorizationEmailHtml(fullAuthData)

+          title: subject,
+          patientName: surgery.patient || "ÔÇö",

+          surgeryDate: surgery.date,
+          procedure: surgery.procedure,
+          companyName,
+          bodyText: notes,
+        })
+
+  // Dispatch Email

+    if (toEmails.length === 0) {

+      return
+    }
+
+    if (!subject.trim()) {
+      toast.error("El asunto no puede estar vac├¡o")

+
